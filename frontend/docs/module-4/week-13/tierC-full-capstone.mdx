        """Handle voice command with safety validation."""
        with self.safety_lock:
            # Validate voice command content
            command_text = msg.data.lower()
            self.voice_cmd_valid = self.validate_voice_command(command_text)

            if self.voice_cmd_valid:

                self.get_logger().debug(f'Valid voice command received: {command_text}')
            else:
                self.get_logger().warn(f'Invalid voice command blocked: {command_text}')

    def validate_voice_command(self, command: str) -> bool:
        """Validate voice command for safety."""
        # Block potentially dangerous commands
        dangerous_keywords = [
            'maximum speed', 'fast', 'quick', 'danger', 'unsafe',
            'ignore', 'bypass', 'override', 'disable safety'
        ]

        for keyword in dangerous_keywords:
            if keyword in command:
                return False

        # Allow only safe navigation commands
        safe_keywords = ['go to', 'move to', 'navigate', 'stop', 'return home']
        return any(keyword in command for keyword in safe_keywords)

        """Handle LLM intent with safety validation."""
        with self.safety_lock:
            intent = msg.data.lower()

            # Validate LLM-generated intent
            self.llm_intent_safe = self.validate_llm_intent(intent)

                self.get_logger().debug(f'Valid LLM intent received: {intent}')

            'disable safety', 'bypass', 'maximum velocity'
        ]

            if pattern in intent:
                return False

        return True

        """Handle navigation goal with safety validation."""
        with self.safety_lock:
            goal_x = msg.pose.position.x
            goal_y = msg.pose.position.y

            # Check if goal is within safe workspace
            if self.initial_position:
                distance_to_goal = np.sqrt(
                    (goal_x - self.initial_position.x)**2 +
                    (goal_y - self.initial_position.y)**2
                )

                self.in_workspace = distance_to_goal <= self.workspace_boundary

                if not self.in_workspace:
                    self.get_logger().warn(f'Navigation goal outside workspace: {distance_to_goal:.2f}m')

    def cmd_vel_callback(self, msg):
        """Handle velocity commands with safety filtering."""
        with self.safety_lock:
            # Apply velocity limits
            filtered_msg = Twist()
            filtered_msg.linear.x = max(min(msg.linear.x, self.max_lin_vel), -self.max_lin_vel)
            filtered_msg.linear.y = max(min(msg.linear.y, self.max_lin_vel), -self.max_lin_vel)
            filtered_msg.angular.z = max(min(msg.angular.z, self.max_ang_vel), -self.max_ang_vel)

            self.last_cmd_vel = filtered_msg
            self.last_cmd_vel_time = self.get_clock().now()

    def joint_state_callback(self, msg):
        """Monitor joint states for safety."""
        with self.safety_lock:
            self.current_joint_states = msg

    def imu_callback(self, msg):
        """Monitor IMU data for balance safety."""
        with self.safety_lock:
            self.current_imu = msg

            # Convert quaternion to Euler angles (simplified)
            w, x, y, z = msg.orientation.w, msg.orientation.x, msg.orientation.y, msg.orientation.z
            pitch = np.arcsin(2 * (w * y - z * x))
            roll = np.arctan2(2 * (w * x + y * z), 1 - 2 * (x*x + y*y))

            # Check if robot is tilting too much
            tilt_magnitude = np.sqrt(pitch*pitch + roll*roll)
            self.balance_ok = tilt_magnitude < self.balance_threshold

    def odom_callback(self, msg):

        with self.safety_lock:
            self.current_odom = msg

            # Store initial position if not set
            if self.initial_position is None:

from launch import LaunchDescription

    ld.add_action(safety_controller)
    ld.add_action(whisper_node)
    ld.add_action(llm_parser)
    ld.add_action(safety_integrator)

### Common Performance Optimizations

1. **Whisper Model Optimization**:
```bash
# Use smaller Whisper models for real-time performance
export WHISPER_MODEL_SIZE=small  # Instead of large
export WHISPER_CHUNK_DURATION=0.5  # 500ms chunks
export WHISPER_THREADS=4  # Limit CPU usage
```

2. **LLM Integration**:
```python
# Optimize LLM calls for real-time response
llm_config = {
    'temperature': 0.1,  # More deterministic
    'max_tokens': 100,   # Limit response length
    'timeout': 5.0,      # 5-second timeout for safety
}
```

3. **VSLAM Optimization**:
```bash
# Reduce VSLAM computational load
export ORBSLAM_N_FEATURES=800    # Fewer features for speed
export ORBSLAM_SCALE_FACTOR=1.1  # Faster scaling
export ORBSLAM_N_LEVELS=6        # Fewer pyramid levels
```

### Troubleshooting Common Issues

**Problem**: System becomes unresponsive during voice processing
**Solutions**:
- Use smaller Whisper models (tiny or small)
- Increase chunk duration for more stable processing
- Add timeout mechanisms to prevent blocking
- Monitor CPU usage during operation

**Problem**: Navigation fails in dynamic environments
**Solutions**:
- Implement fallback strategies for SLAM tracking loss
- Use multiple sensor fusion for robust localization
- Add relocalization capabilities
- Test extensively in similar environments before deployment

**Problem**: Safety system triggers falsely
**Solutions**:
- Fine-tune balance thresholds for specific environment
- Adjust collision detection sensitivity
- Verify IMU calibration
- Test with different locomotion gaits
