# Chapter 20: RTAB-Map

## Introduction

RTAB-Map (Real-Time Appearance-Based Mapping) is a versatile RGB-D SLAM system that leverages visual and depth information to create detailed 3D maps of indoor environments. Unlike laser-based SLAM systems, RTAB-Map uses cameras and depth sensors to extract visual features and build appearance-based maps. This approach is particularly effective in indoor environments where visual landmarks are abundant and distinctive, making RTAB-Map an excellent choice for applications requiring rich visual mapping and localization.

## Real-World Analogy: The Visual Explorer

Think of RTAB-Map like a person exploring a building with a camera and depth sensor instead of a laser scanner. Just as a person remembers visual landmarks like paintings, doorways, and furniture to navigate and map their surroundings, RTAB-Map remembers visual features and their spatial relationships to build maps. The system can recognize familiar visual landmarks from different viewpoints, even when the robot approaches them from different angles, making it robust to viewpoint changes that would confuse laser-only systems.

## Core Concepts

### 1. Appearance-Based Mapping

RTAB-Map uses visual appearance to recognize places and build maps:

- **Visual Bag-of-Words**: Represents locations using visual word histograms
- **Loop Closure Detection**: Recognizes previously visited locations through visual similarity
- **Visual Feature Extraction**: Extracts SIFT, SURF, ORB, or other features
- **Place Recognition**: Matches current view against database of known places

### 2. Graph Optimization

RTAB-Map employs graph-based optimization for consistent mapping:

- **Node-Link Graph**: Represents robot poses and landmark observations
- **Loop Closure Constraints**: Adds constraints when loops are detected
- **Pose Graph Optimization**: Minimizes trajectory errors through optimization
- **Bundle Adjustment**: Optimizes 3D landmark positions and camera poses

### 3. Memory Management

RTAB-Map implements efficient memory management for long-term operation:

- **Database Compression**: Compresses visual features and map data
- **Forgetting Mechanism**: Removes old memories to manage memory usage
- **Multi-resolution Maps**: Maintains maps at different levels of detail
- **Visual Word Pruning**: Removes infrequently observed visual words

```python
#!/usr/bin/env python3

"""
RTAB-Map Node Implementation
Demonstrates core RTAB-Map concepts and integration
"""

import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image, CameraInfo, Imu
from cv_bridge import CvBridge
from geometry_msgs.msg import PointStamped, PoseStamped
from nav_msgs.msg import Odometry
from visualization_msgs.msg import Marker, MarkerArray
from std_msgs.msg import Header
import cv2
import numpy as np
import math
from typing import List, Dict, Tuple, Optional
import threading
import queue


class RTABMapNode(Node):
    def __init__(self):
        super().__init__('rtabmap_node')

        # CV Bridge for image processing
        self.bridge = CvBridge()

        # Subscribers
        self.rgb_sub = self.create_subscription(
            Image, 'camera/rgb/image_raw', self.rgb_callback, 10
        )
        self.depth_sub = self.create_subscription(
            Image, 'camera/depth/image_raw', self.depth_callback, 10
        )
        self.odom_sub = self.create_subscription(
            Odometry, 'odom', self.odom_callback, 10
        )
        self.imu_sub = self.create_subscription(
            Imu, 'imu', self.imu_callback, 10
        )

        # Publishers
        self.map_pub = self.create_publisher(MarkerArray, 'rtabmap_cloud', 10)
        self.odom_pub = self.create_publisher(Odometry, 'rtabmap_odom', 10)
        self.path_pub = self.create_publisher(PoseStamped, 'rtabmap_path', 10)

        # RTAB-Map state
        self.current_rgb = None
        self.current_depth = None
        self.current_odom = None
        self.current_imu = None

        # Feature detection and matching
        self.feature_detector = cv2.SIFT_create(nfeatures=500)
        self.descriptor_matcher = cv2.BFMatcher()
        self.keypoints_db = []  # Database of keypoints
        self.descriptors_db = []  # Database of descriptors
        self.poses_db = []  # Database of poses
        self.loop_closure_threshold = 0.7  # Threshold for loop closure

        # Map representation
        self.map_points = []  # 3D points in the map
        self.trajectory = []  # Robot trajectory

        # Robot state
        self.robot_x = 0.0
        self.robot_y = 0.0
        self.robot_z = 0.0
        self.robot_roll = 0.0
        self.robot_pitch = 0.0
        self.robot_yaw = 0.0

        # Processing queues
        self.image_queue = queue.Queue(maxsize=10)
        self.processing_thread = threading.Thread(target=self.process_images, daemon=True)
        self.processing_thread.start()

        self.get_logger().info('RTAB-Map node initialized')

    def rgb_callback(self, msg):
        """Handle RGB image messages"""
        try:
            cv_image = self.bridge.imgmsg_to_cv2(msg, "bgr8")
            timestamp = msg.header.stamp.sec + msg.header.stamp.nanosec * 1e-9

            # Add to processing queue if not full
            if not self.image_queue.full():
                self.image_queue.put((cv_image, timestamp))
        except Exception as e:
            self.get_logger().error(f'Error processing RGB image: {e}')

    def depth_callback(self, msg):
        """Handle depth image messages"""
        try:
            cv_depth = self.bridge.imgmsg_to_cv2(msg, "32FC1")
            self.current_depth = cv_depth
        except Exception as e:
            self.get_logger().error(f'Error processing depth image: {e}')

    def odom_callback(self, msg):
        """Handle odometry messages"""
        self.current_odom = msg
        self.robot_x = msg.pose.pose.position.x
        self.robot_y = msg.pose.pose.position.y
        self.robot_z = msg.pose.pose.position.z

        # Convert quaternion to Euler angles
        quat = msg.pose.pose.orientation
        self.robot_roll, self.robot_pitch, self.robot_yaw = self.quaternion_to_euler(
            quat.x, quat.y, quat.z, quat.w
        )

    def imu_callback(self, msg):
        """Handle IMU messages for sensor fusion"""
        self.current_imu = msg

    def process_images(self):
        """Process images in separate thread"""
        while rclpy.ok():
            try:
                # Get image from queue
                if not self.image_queue.empty():
                    cv_image, timestamp = self.image_queue.get(timeout=0.1)

                    # Process image for RTAB-Map
                    self.process_rtabmap_frame(cv_image, timestamp)

            except queue.Empty:
                continue
            except Exception as e:
                self.get_logger().error(f'Error in image processing thread: {e}')

    def process_rtabmap_frame(self, cv_image, timestamp):
        """Process a single frame for RTAB-Map"""
        # Extract features from the image
        gray = cv2.cvtColor(cv_image, cv2.COLOR_BGR2GRAY)
        keypoints, descriptors = self.feature_detector.detectAndCompute(gray, None)

        if descriptors is None or len(keypoints) < 10:
            return  # Not enough features to process

        # Store current frame features
        current_keypoints = keypoints
        current_descriptors = descriptors

        # Try to match with previous frames for loop closure
        loop_closure_detected = False
        best_match_idx = -1
        best_match_score = 0.0

        # Compare with stored frames
        for i, (prev_descriptors, prev_pose) in enumerate(zip(self.descriptors_db, self.poses_db)):
            if prev_descriptors is not None:
                # Match descriptors
                matches = self.descriptor_matcher.knnMatch(
                    current_descriptors, prev_descriptors, k=2
                )

                # Apply Lowe's ratio test
                good_matches = []
                for match_pair in matches:
                    if len(match_pair) == 2:
                        m, n = match_pair
                        if m.distance < 0.7 * n.distance:
                            good_matches.append(m)

                # Calculate match score
                if len(good_matches) > 10:  # Minimum matches for consideration
                    match_score = len(good_matches) / max(len(current_descriptors), len(prev_descriptors))
                    if match_score > best_match_score:
                        best_match_score = match_score
                        best_match_idx = i

        # Check for loop closure
        if best_match_score > self.loop_closure_threshold:
            loop_closure_detected = True
            self.get_logger().info(f'Loop closure detected! Match score: {best_match_score:.3f}')

            # Perform graph optimization (simplified)
            self.optimize_graph(best_match_idx)

        # Add current frame to database
        self.keypoints_db.append(current_keypoints)
        self.descriptors_db.append(current_descriptors)

        # Store current pose
        if self.current_odom:
            current_pose = [
                self.current_odom.pose.pose.position.x,
                self.current_odom.pose.pose.position.y,
                self.current_odom.pose.pose.position.z
            ]
        else:
            current_pose = [self.robot_x, self.robot_y, self.robot_z]

        self.poses_db.append(current_pose)

        # Update map with new features (simplified triangulation)
        if self.current_depth is not None:
            self.update_map_from_features(cv_image, current_keypoints, self.current_depth)

        # Publish updated information
        self.publish_map()
        self.publish_odometry()

    def update_map_from_features(self, image, keypoints, depth_image):
        """Update 3D map from features and depth data"""
        # Simplified 3D point extraction using depth at feature locations
        for kp in keypoints[:50]:  # Limit to first 50 features to avoid overload
            u, v = int(kp.pt[0]), int(kp.pt[1])

            # Check bounds
            if 0 <= u < depth_image.shape[1] and 0 <= v < depth_image.shape[0]:
                depth = depth_image[v, u]

                if not (np.isnan(depth) or np.isinf(depth)) and depth > 0:
                    # Convert pixel coordinates to 3D world coordinates (simplified)
                    # In practice, you'd use camera intrinsics for accurate triangulation
                    z_world = depth
                    x_world = (u - image.shape[1]/2) * z_world * 0.001  # Simplified focal length factor
                    y_world = (v - image.shape[0]/2) * z_world * 0.001  # Simplified focal length factor

                    # Transform to global coordinate frame
                    # This is a simplified transformation - in practice, you'd use the full pose
                    global_x = self.robot_x + x_world * math.cos(self.robot_yaw) - y_world * math.sin(self.robot_yaw)
                    global_y = self.robot_y + x_world * math.sin(self.robot_yaw) + y_world * math.cos(self.robot_yaw)
                    global_z = self.robot_z + z_world

                    # Add to map points
                    self.map_points.append([global_x, global_y, global_z])

        # Limit map size to prevent memory issues
        if len(self.map_points) > 10000:
            self.map_points = self.map_points[-5000:]  # Keep last 5000 points

    def optimize_graph(self, loop_idx):
        """Perform simplified graph optimization when loop closure is detected"""
        self.get_logger().info(f'Performing graph optimization with loop to frame {loop_idx}')

        # In a real RTAB-Map implementation, this would use g2o or similar
        # optimization library to adjust poses and minimize drift
        # For this simplified version, we'll just log the event

    def quaternion_to_euler(self, x, y, z, w):
        """Convert quaternion to Euler angles"""
        # Roll (x-axis rotation)
        sinr_cosp = 2 * (w * x + y * z)
        cosr_cosp = 1 - 2 * (x * x + y * y)
        roll = math.atan2(sinr_cosp, cosr_cosp)

        # Pitch (y-axis rotation)
        sinp = 2 * (w * y - z * x)
        if abs(sinp) >= 1:
            pitch = math.copysign(math.pi / 2, sinp)  # Use 90 degrees if out of range
        else:
            pitch = math.asin(sinp)

        # Yaw (z-axis rotation)
        siny_cosp = 2 * (w * z + x * y)
        cosy_cosp = 1 - 2 * (y * y + z * z)
        yaw = math.atan2(siny_cosp, cosy_cosp)

        return roll, pitch, yaw

    def publish_map(self):
        """Publish 3D map as visualization markers"""
        marker_array = MarkerArray()

        # Create markers for map points
        for i, point in enumerate(self.map_points[-500:]):  # Show last 500 points
            marker = Marker()
            marker.header = Header()
            marker.header.stamp = self.get_clock().now().to_msg()
            marker.header.frame_id = 'map'
            marker.ns = 'rtabmap_points'
            marker.id = i
            marker.type = Marker.SPHERE
            marker.action = Marker.ADD

            marker.pose.position.x = point[0]
            marker.pose.position.y = point[1]
            marker.pose.position.z = point[2]
            marker.pose.orientation.w = 1.0

            marker.scale.x = 0.05
            marker.scale.y = 0.05
            marker.scale.z = 0.05

            marker.color.r = 1.0
            marker.color.g = 0.0
            marker.color.b = 0.0
            marker.color.a = 0.8

            marker_array.markers.append(marker)

        self.map_pub.publish(marker_array)

    def publish_odometry(self):
        """Publish RTAB-Map optimized odometry"""
        if self.current_odom:
            odom_msg = Odometry()
            odom_msg.header = Header()
            odom_msg.header.stamp = self.get_clock().now().to_msg()
            odom_msg.header.frame_id = 'map'
            odom_msg.child_frame_id = 'base_link'

            # Use RTAB-Map optimized pose (in this simplified version, same as input)
            odom_msg.pose = self.current_odom.pose
            odom_msg.twist = self.current_odom.twist

            self.odom_pub.publish(odom_msg)

    def get_current_map(self):
        """Get current map for other nodes"""
        return self.map_points

    def get_current_trajectory(self):
        """Get current trajectory"""
        return self.trajectory


def main(args=None):
    rclpy.init(args=args)

    rtabmap_node = RTABMapNode()

    try:
        rclpy.spin(rtabmap_node)
    except KeyboardInterrupt:
        rtabmap_node.get_logger().info('Shutting down RTAB-Map node')
    finally:
        rtabmap_node.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```

```python
#!/usr/bin/env python3

"""
RTAB-Map with IMU Integration
Enhanced RTAB-Map with IMU data fusion for improved pose estimation
"""

import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image, Imu
from geometry_msgs.msg import Vector3
from nav_msgs.msg import Odometry
from std_msgs.msg import Header
import numpy as np
import math
from typing import List, Tuple


class RTABMapIMUIntegration(Node):
    def __init__(self):
        super().__init__('rtabmap_imu_integration')

        # Subscribers
        self.imu_sub = self.create_subscription(
            Imu, 'imu/data', self.imu_callback, 10
        )
        self.odom_sub = self.create_subscription(
            Odometry, 'odom', self.odom_callback, 10
        )

        # Publishers
        self.fused_odom_pub = self.create_publisher(Odometry, 'rtabmap_fused_odom', 10)

        # IMU state estimation
        self.imu_roll = 0.0
        self.imu_pitch = 0.0
        self.imu_yaw = 0.0
        self.imu_linear_acceleration = Vector3()
        self.imu_angular_velocity = Vector3()

        # Odometry state
        self.odom_x = 0.0
        self.odom_y = 0.0
        self.odom_z = 0.0
        self.odom_roll = 0.0
        self.odom_pitch = 0.0
        self.odom_yaw = 0.0

        # Fused state (IMU + Odometry)
        self.fused_x = 0.0
        self.fused_y = 0.0
        self.fused_z = 0.0
        self.fused_roll = 0.0
        self.fused_pitch = 0.0
        self.fused_yaw = 0.0

        # Fusion parameters
        self.imu_weight = 0.7  # Weight for IMU in fusion
        self.odom_weight = 0.3  # Weight for odometry in fusion

        # Timestamps for integration
        self.last_imu_time = None
        self.last_odom_time = None

        self.get_logger().info('RTAB-Map IMU Integration node initialized')

    def imu_callback(self, msg):
        """Handle IMU data for attitude estimation"""
        # Store raw IMU data
        self.imu_linear_acceleration = msg.linear_acceleration
        self.imu_angular_velocity = msg.angular_velocity

        # Convert quaternion to Euler angles
        quat = msg.orientation
        roll, pitch, yaw = self.quaternion_to_euler(quat.x, quat.y, quat.z, quat.w)

        # Apply complementary filter or other fusion technique
        # This is a simplified approach - real implementation would use more sophisticated filtering
        current_time = msg.header.stamp.sec + msg.header.stamp.nanosec * 1e-9

        if self.last_imu_time is not None:
            dt = current_time - self.last_imu_time

            # Integrate angular velocities to get attitude (simplified)
            self.imu_roll += msg.angular_velocity.x * dt
            self.imu_pitch += msg.angular_velocity.y * dt
            self.imu_yaw += msg.angular_velocity.z * dt

        self.last_imu_time = current_time

        # Update fused attitude using weighted average
        self.fused_roll = self.imu_weight * self.imu_roll + self.odom_weight * self.odom_roll
        self.fused_pitch = self.imu_weight * self.imu_pitch + self.odom_weight * self.odom_pitch
        self.fused_yaw = self.imu_weight * self.imu_yaw + self.odom_weight * self.odom_yaw

        # Publish updated fused odometry
        self.publish_fused_odometry()

    def odom_callback(self, msg):
        """Handle odometry data"""
        # Store odometry position
        self.odom_x = msg.pose.pose.position.x
        self.odom_y = msg.pose.pose.position.y
        self.odom_z = msg.pose.pose.position.z

        # Store odometry attitude
        quat = msg.pose.pose.orientation
        self.odom_roll, self.odom_pitch, self.odom_yaw = self.quaternion_to_euler(
            quat.x, quat.y, quat.z, quat.w
        )

        # Update fused position using weighted average
        # In this simplified version, we just use odometry for position
        # but IMU for attitude
        self.fused_x = self.odom_x
        self.fused_y = self.odom_y
        self.fused_z = self.odom_z

        current_time = msg.header.stamp.sec + msg.header.stamp.nanosec * 1e-9
        self.last_odom_time = current_time

        # Publish updated fused odometry
        self.publish_fused_odometry()

    def quaternion_to_euler(self, x, y, z, w):
        """Convert quaternion to Euler angles"""
        # Roll (x-axis rotation)
        sinr_cosp = 2 * (w * x + y * z)
        cosr_cosp = 1 - 2 * (x * x + y * y)
        roll = math.atan2(sinr_cosp, cosr_cosp)

        # Pitch (y-axis rotation)
        sinp = 2 * (w * y - z * x)
        if abs(sinp) >= 1:
            pitch = math.copysign(math.pi / 2, sinp)
        else:
            pitch = math.asin(sinp)

        # Yaw (z-axis rotation)
        siny_cosp = 2 * (w * z + x * y)
        cosy_cosp = 1 - 2 * (y * y + z * z)
        yaw = math.atan2(siny_cosp, cosy_cosp)

        return roll, pitch, yaw

    def publish_fused_odometry(self):
        """Publish fused odometry combining IMU and wheel odometry"""
        odom_msg = Odometry()
        odom_msg.header = Header()
        odom_msg.header.stamp = self.get_clock().now().to_msg()
        odom_msg.header.frame_id = 'map'
        odom_msg.child_frame_id = 'base_link'

        # Set fused position
        odom_msg.pose.pose.position.x = self.fused_x
        odom_msg.pose.pose.position.y = self.fused_y
        odom_msg.pose.pose.position.z = self.fused_z

        # Convert fused Euler angles back to quaternion
        cy = math.cos(self.fused_yaw * 0.5)
        sy = math.sin(self.fused_yaw * 0.5)
        cp = math.cos(self.fused_pitch * 0.5)
        sp = math.sin(self.fused_pitch * 0.5)
        cr = math.cos(self.fused_roll * 0.5)
        sr = math.sin(self.fused_roll * 0.5)

        odom_msg.pose.pose.orientation.w = cr * cp * cy + sr * sp * sy
        odom_msg.pose.pose.orientation.x = sr * cp * cy - cr * sp * sy
        odom_msg.pose.pose.orientation.y = cr * sp * cy + sr * cp * sy
        odom_msg.pose.pose.orientation.z = cr * cp * sy - sr * sp * cy

        # Set covariance (simplified)
        for i in range(36):
            odom_msg.pose.covariance[i] = 0.0
        odom_msg.pose.covariance[0] = 0.1   # x
        odom_msg.pose.covariance[7] = 0.1   # y
        odom_msg.pose.covariance[14] = 0.1  # z
        odom_msg.pose.covariance[21] = 0.05 # rx
        odom_msg.pose.covariance[28] = 0.05 # ry
        odom_msg.pose.covariance[35] = 0.05 # rz

        self.fused_odom_pub.publish(odom_msg)

    def get_fused_state(self):
        """Get current fused state"""
        return (self.fused_x, self.fused_y, self.fused_z,
                self.fused_roll, self.fused_pitch, self.fused_yaw)


def main(args=None):
    rclpy.init(args=args)

    imu_integration = RTABMapIMUIntegration()

        imu_integration.get_logger().info('Shutting down RTAB-Map IMU integration')
    finally:
        imu_integration.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```

```python
#!/usr/bin/env python3

"""
RTAB-Map Map Processor
Processes and optimizes 3D maps from RTAB-Map
"""

import rclpy
from rclpy.node import Node
from sensor_msgs.msg import PointCloud2, PointField
from geometry_msgs.msg import Point
from visualization_msgs.msg import MarkerArray, Marker
from std_msgs.msg import Header
import numpy as np
import math
from typing import List, Tuple
import struct


class RTABMapMapProcessor(Node):
    def __init__(self):
        super().__init__('rtabmap_map_processor')

        # Publishers
        self.processed_map_pub = self.create_publisher(PointCloud2, 'rtabmap_processed_map', 10)
        self.mesh_pub = self.create_publisher(MarkerArray, 'rtabmap_mesh', 10)

        # Map processing parameters
        self.voxel_size = 0.05  # 5cm voxels
        self.distance_threshold = 0.1  # Distance threshold for point clustering
        self.normal_estimation_radius = 0.2  # Radius for normal estimation

        # Processed map data
        self.processed_points = []
        self.map_voxels = {}  # For voxel grid filtering

        # Timer for periodic map processing
        self.process_timer = self.create_timer(5.0, self.process_and_publish_map)

        self.get_logger().info('RTAB-Map Map Processor initialized')

    def add_points_to_map(self, points_3d: List[Tuple[float, float, float]]):
        """Add 3D points to the map with processing"""
        # Voxel grid filtering to reduce point cloud density
        filtered_points = self.voxel_grid_filter(points_3d)

        # Update processed points
        self.processed_points.extend(filtered_points)

        # Limit map size to prevent memory issues
        if len(self.processed_points) > 50000:  # Limit to 50k points
            self.processed_points = self.processed_points[-25000:]  # Keep last 25k points

    def voxel_grid_filter(self, points: List[Tuple[float, float, float]]) -> List[Tuple[float, float, float]]:
        """Apply voxel grid filtering to reduce point density"""
        filtered_points = []
        self.map_voxels.clear()  # Clear voxel grid

        for point in points:
            x, y, z = point

            # Calculate voxel coordinates
            voxel_x = int(x / self.voxel_size)
            voxel_y = int(y / self.voxel_size)
            voxel_z = int(z / self.voxel_size)

            voxel_key = (voxel_x, voxel_y, voxel_z)

            # If this voxel doesn't exist, add the point
            if voxel_key not in self.map_voxels:
                self.map_voxels[voxel_key] = point
                filtered_points.append(point)

        return filtered_points

    def create_pointcloud2(self, points: List[Tuple[float, float, float]]) -> PointCloud2:
        """Create PointCloud2 message from list of 3D points"""
        # Create PointCloud2 message
        cloud = PointCloud2()
        cloud.header = Header()
        cloud.header.stamp = self.get_clock().now().to_msg()
        cloud.header.frame_id = 'map'

        # Define fields
        cloud.fields = [
            PointField(name='x', offset=0, datatype=PointField.FLOAT32, count=1),
            PointField(name='y', offset=4, datatype=PointField.FLOAT32, count=1),
            PointField(name='z', offset=8, datatype=PointField.FLOAT32, count=1)
        ]

        cloud.point_step = 12  # 3 floats * 4 bytes each
        cloud.row_step = cloud.point_step * len(points)
        cloud.is_dense = True

        # Pack points into binary data
        data = []
        for point in points:
            data.append(struct.pack('fff', point[0], point[1], point[2]))

        cloud.data = b''.join(data)
        cloud.width = len(points)
        cloud.height = 1

        return cloud

    def create_mesh_markers(self, points: List[Tuple[float, float, float]]) -> MarkerArray:
        """Create mesh markers for visualization"""
        marker_array = MarkerArray()

        # Create surface mesh approximation (simplified)
        if len(points) > 3:
            # Group nearby points to create surface patches
            for i in range(0, min(len(points), 1000), 100):  # Sample points to avoid too many markers
                if i + 2 < len(points):
                    # Create a triangle marker
                    marker = Marker()
                    marker.header = Header()
                    marker.header.stamp = self.get_clock().now().to_msg()
                    marker.header.frame_id = 'map'
                    marker.ns = 'rtabmap_mesh'
                    marker.id = i // 100
                    marker.type = Marker.TRIANGLE_LIST
                    marker.action = Marker.ADD

                    marker.pose.orientation.w = 1.0
                    marker.scale.x = 1.0
                    marker.scale.y = 1.0
                    marker.scale.z = 1.0

                    marker.color.r = 0.0
                    marker.color.g = 0.5
                    marker.color.b = 1.0
                    marker.color.a = 0.3

                    # Add three points to form a triangle
                    p1 = Point()
                    p1.x, p1.y, p1.z = points[i]
                    marker.points.append(p1)

                    p2 = Point()
                    p2.x, p2.y, p2.z = points[min(i+1, len(points)-1)]
                    marker.points.append(p2)

                    p3 = Point()
                    p3.x, p3.y, p3.z = points[min(i+2, len(points)-1)]
                    marker.points.append(p3)

                    marker_array.markers.append(marker)

        return marker_array

    def process_and_publish_map(self):
        """Process and publish the 3D map"""
        if not self.processed_points:
            return

        # Create and publish point cloud
        pointcloud_msg = self.create_pointcloud2(self.processed_points)
        self.processed_map_pub.publish(pointcloud_msg)

        # Create and publish mesh markers
        mesh_markers = self.create_mesh_markers(self.processed_points)
        self.mesh_pub.publish(mesh_markers)

        self.get_logger().info(f'Published processed map with {len(self.processed_points)} points')

    def get_processed_map(self) -> List[Tuple[float, float, float]]:
        """Get the processed map points"""
        return self.processed_points.copy()

    def clear_map(self):
        """Clear the current map"""
        self.processed_points.clear()
        self.map_voxels.clear()
        self.get_logger().info('RTAB-Map cleared')


def main(args=None):
    rclpy.init(args=args)

    map_processor = RTABMapMapProcessor()

    # Example: Add some sample points to the map
    def add_sample_points():
        # Add sample points in a grid pattern
        sample_points = []
        for x in np.arange(-2.0, 2.0, 0.2):
            for y in np.arange(-2.0, 2.0, 0.2):
                z = math.sin(x) * math.cos(y)  # Create a wavy surface
                sample_points.append((x, y, z))

        map_processor.add_points_to_map(sample_points)
        map_processor.get_logger().info(f'Added {len(sample_points)} sample points to map')

    # Add sample points after a delay
    timer = map_processor.create_timer(2.0, add_sample_points)

    try:
        rclpy.spin(map_processor)
    except KeyboardInterrupt:
        map_processor.get_logger().info('Shutting down RTAB-Map processor')
    finally:
        map_processor.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```

```python
#!/usr/bin/env python3

"""
RTAB-Map Performance Monitor
Monitors and reports RTAB-Map performance metrics
"""

import rclpy
from rclpy.node import Node
from std_msgs.msg import Float64, Int32, String
from geometry_msgs.msg import PoseWithCovarianceStamped
import time
import numpy as np
from typing import List


class RTABMapPerformanceMonitor(Node):
    def __init__(self):
        super().__init__('rtabmap_performance_monitor')

        # Subscribers
        self.pose_sub = self.create_subscription(
            PoseWithCovarianceStamped, 'rtabmap_pose', self.pose_callback, 10
        )

        # Publishers
        self.fps_pub = self.create_publisher(Float64, 'rtabmap_fps', 10)
        self.map_size_pub = self.create_publisher(Int32, 'rtabmap_map_size', 10)
        self.processing_time_pub = self.create_publisher(Float64, 'rtabmap_processing_time', 10)
        self.status_pub = self.create_publisher(String, 'rtabmap_status', 10)

        # Performance tracking
        self.frame_times = []
        self.map_sizes = []
        self.processing_times = []
        self.last_pose_time = None

        # Metrics
        self.average_fps = 0.0
        self.current_map_size = 0
        self.average_processing_time = 0.0

        # Timer for metrics calculation
        self.metrics_timer = self.create_timer(1.0, self.calculate_metrics)

        self.get_logger().info('RTAB-Map Performance Monitor initialized')

    def pose_callback(self, msg):
        """Handle pose messages to track processing rate"""
        current_time = time.time()

        if self.last_pose_time is not None:
            frame_time = current_time - self.last_pose_time
            self.frame_times.append(frame_time)

            # Limit history to last 30 measurements
            if len(self.frame_times) > 30:
                self.frame_times = self.frame_times[-30:]

        self.last_pose_time = current_time

    def calculate_metrics(self):
        """Calculate and publish performance metrics"""
        # Calculate FPS
        if self.frame_times:
            avg_frame_time = np.mean(self.frame_times)
            self.average_fps = 1.0 / avg_frame_time if avg_frame_time > 0 else 0.0
        else:
            self.average_fps = 0.0

        # Publish FPS
        fps_msg = Float64()
        fps_msg.data = float(self.average_fps)
        self.fps_pub.publish(fps_msg)

        # Calculate and publish map size (simulated)
        self.current_map_size = len(self.frame_times) * 10  # Simulated map size
        map_size_msg = Int32()
        map_size_msg.data = self.current_map_size
        self.map_size_pub.publish(map_size_msg)

        # Calculate average processing time (simulated)
        if self.processing_times:
            self.average_processing_time = np.mean(self.processing_times[-10:])
        else:
            self.average_processing_time = 0.0

        proc_time_msg = Float64()
        proc_time_msg.data = float(self.average_processing_time)
        self.processing_time_pub.publish(proc_time_msg)

        # Create status message
        status_msg = String()
        status_msg.data = f"FPS: {self.average_fps:.1f}, Map: {self.current_map_size}, Proc: {self.average_processing_time:.3f}s"
        self.status_pub.publish(status_msg)

        # Log performance
        self.get_logger().info(f'RTAB-Map Performance - FPS: {self.average_fps:.1f}, Map size: {self.current_map_size}')

    def get_current_metrics(self):
        """Get current performance metrics"""
        return {
            'fps': self.average_fps,
            'map_size': self.current_map_size,
            'processing_time': self.average_processing_time
        }


def main(args=None):
    rclpy.init(args=args)

    monitor = RTABMapPerformanceMonitor()

    try:
        rclpy.spin(monitor)
    except KeyboardInterrupt:
        monitor.get_logger().info('Shutting down RTAB-Map performance monitor')
    finally:
        monitor.destroy_node()
        rclpy.shutdown()


if __name__ == '__main__':
    main()
```
